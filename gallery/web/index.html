pub fn classify(width: Fixed, height: Fixed) -> Self {
    let short = width.min(height);
    let long = width.max(height);
    if long * 10 <= short * 12 && long <= Fixed::from_int(480) {
        Self::Dial
    } else if short < Fixed::from_int(200) {
        Self::Micro
    } else if height * 10 > width * 13 {
        Self::Portrait
    } else {
        Self::Landscape
    }
}

How it works

A retained component tree, laid out in Q24.8 fixed point, drawn by the backend the target has.

main.rsREADME.md
#[compose]
fn build_root() {
    ui! {
        Column (grow: 1.0) {
            View (
                bg_color: ColorToken::Primary,
                text_color: ColorToken::OnPrimary,
                height: 40,
                border_radius: 8,
                padding: Padding::all(10)
            ) {
                Text ("Hello mirui!")
            }
            View (bg_color: ColorToken::SurfaceVariant, grow: 1.0)
            Text ("ECS + DSL", height: 30, paragraph: ParagraphStyle::label())
        }
    };
}
  • #[compose] functions return reusable subtrees
  • walk loops and if / match branches inside the tree
  • Integer literals coerce to Fixed and Dimension
counter.rsREADME.md · value 41
#[model]
struct Counter {
    #[observe]
    value: i32,
}

#[compose]
fn counter_label(counter: model!(Counter), prefix: &'static str) -> Entity {
    ui! { Text(text: ${ format!("{prefix}: {}", counter.value()) }) }
}

#[compose]
fn counter_panel(counter: model!(Counter)) {
    ui! {
        Row {
            Button(text: "+") on Tap { counter.increment(); }
            counter_label(counter, "count")
        }
    };
}

// One tap on + and the label reads 42.
let counter = app.add_model(Counter { value: 41 }).finish();
  • $ subscribes to the fields a binding reads
  • Errors from a method publish no updates
  • One model can feed many views

main.rsREADME.md
fn main() {
    let backend = SdlSurface::new("hello mirui", 480, 320);
    let mut app = App::new(backend);
    app.with_default_widgets().with_default_systems();

    // Root fills the viewport with a Surface bg + Column layout by default.
    let root = app.spawn_root().id();
    app.compose(root, build_root());

    app.run();
}
